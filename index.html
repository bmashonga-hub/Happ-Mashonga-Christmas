// HOST SIGN-IN async function hostSignIn() { const email = prompt("Enter your host email address:"); if (!email) return; const password = prompt("Enter your Supabase password:"); if (!password) return; const { data, error } = await db.auth.signInWithPassword({ email, password }); if (error) { alert("Sign-in failed. Please check your details."); return; } const { data: isHost, error: hostError } = await db.rpc("check_host_access"); if (hostError || !isHost) { await db.auth.signOut(); alert("This account is not authorised as the host."); return; } alert("Welcome, host! You are signed in."); }
